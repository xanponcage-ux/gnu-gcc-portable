(
    SELECT
        ROUND(SUM(NVL(EOP_BTD,0)),0) EOP_BTD,
        ROUND(SUM(NVL(EOP_BTP,0)),0) EOP_BTP,
        ROUND(SUM(NVL(EOP_MILLSCH_QTY,0)),0) EOP_MILLSCH_QTY,
        ROUND(SUM(NVL(EOP_SIGNOF_QTY,0)),0) EOP_SIGNOF_QTY,
        ROUND(SUM(NVL(EOP_VDI_GRN,0)),0) GRNCOMP_QTY,
        ROUND(
       NVL(
            (SUM(NVL(EOP_VDI_GRN,0))
             / NULLIF(SUM(NVL(EOP_SIGNOF_QTY,0)),0)
            ) * 100,
       0)
     ,0) AS GRNCOMP_PER
        FROM V_LDP_ORDER_PROG,
         V_END_CUST_ORD_EPA
    WHERE EOP_CD_EPA = ENC_CD_EPA
      AND EOP_ID_ORDER = ENC_ID_ORDER
      AND EOP_NO_ITEM = ENC_NO_ITEM
      AND ENC_DT_ORD_CREATE > SYSDATE - 365
      AND ENC_ST_ORDER = 'A'
      AND EOP_CD_EPA = '0780'
) M
CROSS JOIN
(
    SELECT
        -- Segment-based sums
        ROUND(SUM(CASE WHEN EOP_SEGEMENT = 'Branded Products & Retail' THEN NVL(EOP_BTD,0) ELSE 0 END),0) AS BPRTUBE_BTD,
        ROUND(SUM(CASE WHEN EOP_SEGEMENT = 'Branded Products & Retail' THEN NVL(EOP_BTP,0) ELSE 0 END),0) AS BPRTUBE_BTP,
        ROUND(SUM(CASE WHEN EOP_SEGEMENT = 'Commercial & Exports' THEN NVL(EOP_BTD,0) ELSE 0 END),0) AS CETUBE_BTD,
        ROUND(SUM(CASE WHEN EOP_SEGEMENT = 'Commercial & Exports' THEN NVL(EOP_BTP,0) ELSE 0 END),0) AS CETUBE_BTP,

        ROUND(SUM(CASE WHEN EOP_SEGEMENT = 'Construction and Solution' THEN NVL(EOP_BTP,0) ELSE 0 END),0) AS CSTUBE_BTP,

        ROUND(SUM(CASE WHEN EOP_SEGEMENT = 'Engineering & Packaging - Tubes' THEN NVL(EOP_BTP,0) ELSE 0 END),0) AS EPTUBE_BTP,
        -- Zone-based sums
        ROUND(SUM(CASE WHEN EOP_ZONE = 'API' THEN NVL(EOP_BTD,0) ELSE 0 END),0) AS API_BTD,
        ROUND(SUM(CASE WHEN EOP_ZONE = 'API' THEN NVL(EOP_BTP,0) ELSE 0 END),0) AS API_BTP,
        ROUND(SUM(CASE WHEN EOP_ZONE = 'East' THEN NVL(EOP_BTD,0) ELSE 0 END),0) AS EAST_BTD,
        ROUND(SUM(CASE WHEN EOP_ZONE = 'East' THEN NVL(EOP_BTP,0) ELSE 0 END),0) AS EAST_BTP,
        ROUND(SUM(CASE WHEN EOP_ZONE = 'North' THEN NVL(EOP_BTD,0) ELSE 0 END),0) AS NORTH_BTD,
        ROUND(SUM(CASE WHEN EOP_ZONE = 'North' THEN NVL(EOP_BTP,0) ELSE 0 END),0) AS NORTH_BTP,
        ROUND(SUM(CASE WHEN EOP_ZONE = 'South' THEN NVL(EOP_BTD,0) ELSE 0 END),0) AS SOUTH_BTD,
        ROUND(SUM(CASE WHEN EOP_ZONE = 'South' THEN NVL(EOP_BTP,0) ELSE 0 END),0) AS SOUTH_BTP,
        ROUND(SUM(CASE WHEN EOP_ZONE = 'West' THEN NVL(EOP_BTD,0) ELSE 0 END),0) AS WEST_BTD,
        ROUND(SUM(CASE WHEN EOP_ZONE = 'West' THEN NVL(EOP_BTP,0) ELSE 0 END),0) AS WEST_BTP,
        ROUND(SUM(CASE WHEN EOP_ZONE = 'Export' THEN NVL(EOP_BTD,0) ELSE 0 END),0) AS EXPORT_BTD,
        ROUND(SUM(CASE WHEN EOP_ZONE = 'Export' THEN NVL(EOP_BTP,0) ELSE 0 END),0) AS EXPORT_BTP
    FROM V_LDP_ORDER_PROG
